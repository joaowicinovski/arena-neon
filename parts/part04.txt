  function resumeGame(){
    if(state === 'pause'){
      state = 'playing';
      menuOverlay.style.display = 'none';
    }
  }

  function togglePause(){
    if(state === 'playing') showPause();
    else if(state === 'pause') resumeGame();
  }

  function floatingText(x,y,text,color='white'){

  }

  function particles(x,y,color,count=12,force=1){
    for(let i=0;i<count;i++){
      const a = Math.random()*Math.PI*2;
      const s = random(45,220)*force;
      game.particles.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,r:random(1.4,4.4),life:random(.35,.85),color});
    }
  }
