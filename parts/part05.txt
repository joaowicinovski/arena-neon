  function addDrone(){
    const i = game.drones.length;

  }

  function shootFrom(x,y,angle,owner='player',damage=null,speed=null,pierce=null,color=null){
    const p = game.player;
    const isPlayer = owner === 'player';
    const crit = isPlayer && Math.random() < p.crit;
    game.bullets.push({
      x,y,
      vx:Math.cos(angle)*(speed ?? (isPlayer ? p.bulletSpeed : 300)),
      vy:Math.sin(angle)*(speed ?? (isPlayer ? p.bulletSpeed : 300)),
      r:isPlayer?4.4:5.2,
      life:isPlayer?1.25:3.2,
      owner,
      damage:(damage ?? (isPlayer ? p.damage : 12))*(crit?2:1),
      pierce:pierce ?? (isPlayer ? p.pierce : 0),
      color:color || (crit ? '#ffe06a' : (isPlayer ? '#58e7ff' : '#5ba7ff'))
    });
    if(crit) floatingText(x,y,tr('crit'),'#ffe06a');
  }

  function getAimAngle(){
    const p = game.player;
    if(!p) return 0;
    return touch.aimId !== null && touch.aimAngle !== null ? touch.aimAngle : angleTo(p.x,p.y,mouse.x,mouse.y);
  }

  function playerShoot(now){
    const p = game.player;
    const rapid = game.rapidUntil > game.time ? .48 : 1;
    if(now - p.lastShot < p.fireDelay * rapid) return;
    p.lastShot = now;
    const a = getAimAngle();
    shootFrom(p.x+Math.cos(a)*p.r,p.y+Math.sin(a)*p.r,a);
    const extra = Math.min(4,p.spread);
    for(let i=1;i<=extra;i++){
      const off = .12*i;
      shootFrom(p.x,p.y,a+off,'player',p.damage*.78,p.bulletSpeed*.94,p.pierce);
      shootFrom(p.x,p.y,a-off,'player',p.damage*.78,p.bulletSpeed*.94,p.pierce);
    }
  }

  function registerUpgrade(u){
    if(!game.upgrades[u.id]) game.upgrades[u.id] = {icon:u.icon,nameKey:u.nameKey,count:0};
    game.upgrades[u.id].count++;
    updateUpgradeList();
  }

  function updateUpgradeList(){
    const items = Object.values(game.upgrades);
    if(items.length === 0){
      upgradesList.innerHTML = `<span class="empty">${tr('none')}</span>`;
      return;
    }
    upgradesList.innerHTML = items.map(item => `<span class="chip"><span>${item.icon}</span><span>${tr(item.nameKey).replace(/\s*\+.*/,'')}</span><span class="count">x${item.count}</span></span>`).join('');
  }
