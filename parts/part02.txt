  function serializeProgress(){
    return JSON.stringify({version:1,best:Math.floor(game.best),settings:{optionCount:config.optionCount,pickCount:config.pickCount,autoFire:!!config.autoFire}});
  }

  function applySavedProgress(raw){
    if(!raw) return;
    try{
      const data=JSON.parse(raw);
      if(data && Number.isFinite(Number(data.best))) game.best=Math.max(0,Number(data.best));
      if(data && data.settings){
        const s=data.settings;
        config.optionCount=clamp(parseInt(s.optionCount,10)||3,1,upgradePool.length);
        config.pickCount=clamp(parseInt(s.pickCount,10)||1,1,Math.max(1,config.optionCount));
        config.autoFire=!!s.autoFire;
      }
    }catch(_){ /* Ignore incompatible/corrupt older saves. */ }
  }

  async function saveProgress(){
    if(!platform.loaded) return false;
    const data=serializeProgress();
    try{
      if(platform.inPlayables){
        await ytgame.game.saveData(data);
      }else{
        localStorage.setItem('arena_neon_save',data);
      }
      return true;
    }catch(_){ return false; }
  }

  function queueSave(){
    if(saveQueued || !platform.loaded) return;
    saveQueued=true;
    Promise.resolve().then(async()=>{saveQueued=false;await saveProgress();});
  }

  async function sendBestScore(force=false){
    if(!platform.inPlayables || !platform.loaded) return;
    const score=Math.floor(game.best);
    if(score<=platform.lastSentScore && !force) return;
    const saved=await saveProgress();
    if(!saved) return;
    try{await ytgame.engagement.sendScore({value:score});platform.lastSentScore=Math.max(platform.lastSentScore,score);}catch(_){ }
  }

  function onPlatformPause(){
    if(platform.paused) return;
    platform.paused=true;
    mouse.down=false;
    clearTouchControls();
    if(rafId){cancelAnimationFrame(rafId);rafId=0;}
    void saveProgress();
  }

  function onPlatformResume(){
    if(!platform.paused) return;
    platform.paused=false;
    lastTime=performance.now();
    startLoop();
  }
