    const g = ctx.createRadialGradient(W*.5,H*.45,20,W*.5,H*.5,Math.max(W,H)*.75);
    g.addColorStop(0,'rgba(28,53,101,.72)');
    g.addColorStop(.55,'rgba(7,12,28,.86)');
    g.addColorStop(1,'rgba(1,3,8,1)');
    ctx.fillStyle = g;
    ctx.fillRect(0,0,W,H);
    for(const s of stars){
      s.y += s.s*.2;
      if(s.y > H) s.y = 0;
      ctx.globalAlpha = s.a;
      ctx.fillStyle = '#dff8ff';
      ctx.beginPath();ctx.arc(s.x,s.y,s.r,0,Math.PI*2);ctx.fill();
    }
    ctx.globalAlpha = 1;
    ctx.strokeStyle = 'rgba(88,231,255,.10)';
    ctx.lineWidth = 1;
    const gap = 48;
    const ox = (game.time*10)%gap;
    const oy = (game.time*6)%gap;
    for(let x=-gap+ox;x<W+gap;x+=gap){ctx.beginPath();ctx.moveTo(x,0);ctx.lineTo(x,H);ctx.stroke();}
    for(let y=-gap+oy;y<H+gap;y+=gap){ctx.beginPath();ctx.moveTo(0,y);ctx.lineTo(W,y);ctx.stroke();}
  }

  function draw(){
    ctx.clearRect(0,0,W,H);
    ctx.save();
    if(shake>0) ctx.translate(random(-shake,shake),random(-shake,shake));
    drawBackground();

    ctx.save();
    ctx.globalAlpha=.92;
    ctx.fillStyle='rgba(4,8,18,.58)';
    ctx.fillRect(12,12,Math.min(252,W-24),58);
    ctx.fillStyle='#edf7ff';ctx.font='bold 14px Arial';ctx.textAlign='left';
    ctx.fillText(`${tr('score')}: ${Math.floor(game.score)}   ${tr('wave')}: ${game.wave}   ${tr('level')}: ${game.level}`,22,34);
    if(game.player){
      const hp=clamp(game.player.hp/game.player.maxHp,0,1);
      ctx.fillStyle='rgba(255,255,255,.13)';ctx.fillRect(22,45,200,10);
      ctx.fillStyle='#70ff9d';ctx.fillRect(22,45,200*hp,10);
    }
    ctx.restore();

    for(const h of game.hazards){
      const pulse = Math.sin(h.pulse*8)*4;
      ctx.save();ctx.globalAlpha=clamp(h.life/2,0,1)*.65;ctx.fillStyle='rgba(255,85,115,.13)';ctx.strokeStyle='rgba(255,85,115,.55)';ctx.lineWidth=2;ctx.beginPath();ctx.arc(h.x,h.y,h.r+pulse,0,Math.PI*2);ctx.fill();ctx.stroke();ctx.restore();
    }

    for(const item of game.pickups){
      const bob = Math.sin(game.time*6 + item.x)*3;
      ctx.save();ctx.translate(item.x,item.y+bob);ctx.rotate(game.time*2);ctx.shadowBlur=18;ctx.shadowColor=item.color;ctx.fillStyle=item.color;
      if(item.type==='crystal'){ctx.beginPath();ctx.moveTo(0,-item.r);ctx.lineTo(item.r,0);ctx.lineTo(0,item.r);ctx.lineTo(-item.r,0);ctx.closePath();ctx.fill();}
      else {ctx.beginPath();ctx.arc(0,0,item.r,0,Math.PI*2);ctx.fill();ctx.fillStyle='rgba(0,0,0,.28)';ctx.beginPath();ctx.arc(0,0,item.r*.45,0,Math.PI*2);ctx.fill();}
      ctx.restore();
    }

    for(const b of game.bullets){
      ctx.save();ctx.shadowBlur=16;ctx.shadowColor=b.color;ctx.fillStyle=b.color;ctx.beginPath();ctx.arc(b.x,b.y,b.r,0,Math.PI*2);ctx.fill();ctx.restore();
    }

    for(const e of game.enemies){
      ctx.save();ctx.shadowBlur=22;ctx.shadowColor=e.color;ctx.fillStyle=e.hitFlash>0?'#fff':e.color;
      if(e.kind==='tank'){
        ctx.beginPath();
        for(let i=0;i<6;i++){const a=i*Math.PI/3+game.time*.5;const rr=e.r*(i%2?.86:1.05);const x=e.x+Math.cos(a)*rr;const y=e.y+Math.sin(a)*rr;if(i===0)ctx.moveTo(x,y);else ctx.lineTo(x,y);}
        ctx.closePath();ctx.fill();
      } else if(e.kind==='runner'){
        const a = angleTo(e.x,e.y,game.player?.x||W/2,game.player?.y||H/2);
        ctx.translate(e.x,e.y);ctx.rotate(a);ctx.beginPath();ctx.moveTo(e.r+5,0);ctx.lineTo(-e.r,e.r*.75);ctx.lineTo(-e.r,-e.r*.75);ctx.closePath();ctx.fill();ctx.setTransform(DPR,0,0,DPR,0,0);
      } else {ctx.beginPath();ctx.arc(e.x,e.y,e.r,0,Math.PI*2);ctx.fill();}
      ctx.shadowBlur=0;ctx.fillStyle='rgba(0,0,0,.46)';const w=e.r*2.1;ctx.fillRect(e.x-w/2,e.y-e.r-12,w,4);ctx.fillStyle='rgba(255,255,255,.86)';ctx.fillRect(e.x-w/2,e.y-e.r-12,w*clamp(e.hp/e.maxHp,0,1),4);ctx.restore();
    }

    const p = game.player;
    if(p){
      const a = getAimAngle();
      ctx.save();ctx.globalAlpha=.46;ctx.strokeStyle='#58e7ff';ctx.lineWidth=2;ctx.setLineDash([8,10]);ctx.beginPath();ctx.moveTo(p.x,p.y);ctx.lineTo(p.x+Math.cos(a)*90,p.y+Math.sin(a)*90);ctx.stroke();ctx.restore();
      if(game.shieldUntil > game.time){ctx.save();ctx.strokeStyle='rgba(155,124,255,.78)';ctx.lineWidth=4;ctx.shadowBlur=24;ctx.shadowColor='#9b7cff';ctx.beginPath();ctx.arc(p.x,p.y,p.r+10+Math.sin(game.time*8)*2,0,Math.PI*2);ctx.stroke();ctx.restore();}
      ctx.save();ctx.translate(p.x,p.y);ctx.rotate(a);ctx.shadowBlur=28;ctx.shadowColor='#58e7ff';ctx.fillStyle=p.invuln>0 && Math.floor(game.time*18)%2===0?'rgba(255,255,255,.45)':'#58e7ff';ctx.beginPath();ctx.moveTo(p.r+8,0);ctx.lineTo(-p.r,p.r*.8);ctx.lineTo(-p.r*.45,0);ctx.lineTo(-p.r,-p.r*.8);ctx.closePath();ctx.fill();ctx.fillStyle='rgba(255,255,255,.86)';ctx.beginPath();ctx.arc(-3,0,4,0,Math.PI*2);ctx.fill();ctx.restore();
      for(const d of game.drones){ctx.save();ctx.shadowBlur=18;ctx.shadowColor='#70ff9d';ctx.fillStyle='#70ff9d';ctx.beginPath();ctx.arc(d.x,d.y,7,0,Math.PI*2);ctx.fill();ctx.restore();}
    }

    for(const q of game.particles){ctx.save();ctx.globalAlpha=clamp(q.life,0,1);ctx.fillStyle=q.color;ctx.beginPath();ctx.arc(q.x,q.y,q.r,0,Math.PI*2);ctx.fill();ctx.restore();}
    for(const t of game.texts){ctx.save();ctx.globalAlpha=clamp(t.life/.85,0,1);ctx.fillStyle=t.color;ctx.font='bold 16px Arial';ctx.textAlign='center';ctx.fillText(t.text,t.x,t.y);ctx.restore();}
    ctx.restore();
  }

  function loop(now){
    rafId = 0;
    if(platform.paused) return;
    const dt = Math.min(.033,(now-lastTime)/1000 || .016);
    lastTime = now;
    update(dt,now);
    draw();
    rafId = requestAnimationFrame(loop);
  }

  function canvasPos(evt){
    const r = canvas.getBoundingClientRect();
    return {x:(evt.clientX-r.left)*W/r.width,y:(evt.clientY-r.top)*H/r.height};
  }

  function moveStickVisual(knob,dx,dy){ if(knob) knob.style.transform=`translate(calc(-50% + ${dx*34}px),calc(-50% + ${dy*34}px))`; }

    touch.moveId=null;touch.aimId=null;touch.moveX=0;touch.moveY=0;touch.aimAngle=null;mouse.down=false;
    resetStickVisual(moveKnob);resetStickVisual(aimKnob);
  }
